================================================================================
AWS - USAGE
================================================================================

CONTROLLER

SSH:

ssh -i my-key.pem ubuntu@<CONTROLLER_PUBLIC_IP>

Go to scripts:

cd ~/self-managed-k8s-automation/Aws

Make executable:

chmod +x controller.sh

Run:

sudo ./controller.sh <CONTROLLER_PRIVATE_IP>


INSTALLED / CONFIGURED

- containerd
- kubeadm
- kubelet
- kubectl
- Kubernetes Control Plane
- CNI Plugin


GET JOIN COMMAND

kubeadm token create --print-join-command


WORKER

SSH:

ssh -i my-key.pem ubuntu@<WORKER_PUBLIC_IP>

Go to scripts:

cd ~/self-managed-k8s-automation/Aws

Make executable:

chmod +x worker.sh

Run:

sudo ./worker.sh <CONTROLLER_PRIVATE_IP:6443> <TOKEN> <CA_CERT_HASH>


INSTALLED / CONFIGURED

- containerd
- kubeadm
- kubelet
- Kubernetes Worker Node
- Kubernetes CNI networking


VERIFY CLUSTER

kubectl get nodes -o wide

kubectl get pods -A

kubectl get svc -A

kubectl get deployments -A


RESET WORKER

sudo kubeadm reset -f
